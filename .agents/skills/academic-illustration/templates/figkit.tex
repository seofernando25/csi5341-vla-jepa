% Shared standard for all TikZ explainer figures (include after \documentclass[tikz,border=6pt]{standalone}).
% Rules: final width 16.5 cm (= \linewidth, scale 1.0); body text 8 pt, labels >= 7 pt, never smaller;
% every arrow starts/ends on a node anchor (never floats over text); edge labels use `fill=white` or sit
% above/below the arrow with at least 2 mm clearance; every box has inner sep >= 4 pt and its text fits
% inside (use text width / align=left / manual line breaks, then check the render); >= 5 mm gap between boxes;
% >= 4 mm padding between content and panel border; one accent colour per meaning (garnet = mismatch/bug).

% \StagePanel{name}{x}{y top}{width cm}{height cm}{base|derive|variant}{step}{title}{subtitle}
% Defines (name-in): the top centre of the content area, below the subtitle.
\newcommand{\StagePanel}[9]{%
  \pgfmathsetlengthmacro{\stagesubw}{(#4-0.6)*1cm}%
  \node[stagebox=#6, minimum width=#4cm, minimum height=#5cm, anchor=north west]
    (#1) at (#2,#3) {};
  \node[stepbadge] (#1-n) at ($(#1.north west)+(0.48,-0.42)$) {#7};
  \node[stagetitle, anchor=west] at ($(#1-n.east)+(0.13,0)$) {#8};
  \node[stagesub, anchor=north west, text width=\stagesubw] (#1-s)
    at ($(#1.north west)+(0.3,-0.72)$) {#9};
  \coordinate (#1-in) at ($(#1.north |- #1-s.south)+(0,-0.22)$);
}

% \CompareFooter{left outcome}{right outcome}{bug text}: one bus, one decision, two pills.
\newcommand{\CompareFooter}[3]{%
  \coordinate (busL) at ($(#1.south)+(0,-0.42)$);
  \coordinate (busR) at (#2.south |- busL);
  \coordinate (busM) at ($(busL)!0.5!(busR)$);
  \draw[busline] (#1.south) -- (busL) -- (busR) -- (#2.south);
  \node[dia] (eq) at ($(busM)+(0,-0.78)$) {equal?};
  \draw[flow] (busM) -- (eq.north);
  \node[pill, draw=okgreen, fill=pgreen, text=okgreen] (fine) at ($(eq)+(-3.3,-0.95)$)
    {\faCheck\enspace fine};
  \node[pill, draw=garnet, fill=pgarnet, text=garnet] (bug) at ($(eq)+(3.3,-0.95)$)
    {\faBug\enspace #3};
  \draw[flow, okgreen, line width=.8pt] (eq.west) -| (fine.north);
  \draw[flow, garnet, line width=.8pt] (eq.east) -| (bug.north);
  \node[elab, text=okgreen, anchor=south] at ($(eq.west)!0.5!(eq.west -| fine.north)$) {yes};
  \node[elab, text=garnet, anchor=south] at ($(eq.east)!0.5!(eq.east -| bug.north)$) {no};
}
